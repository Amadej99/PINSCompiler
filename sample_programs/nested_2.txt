        { v1 = 1 },

        f2(20),
        0
    ) { where
        fun f2(p2 : integer) : integer = (
            { v2 = 2 },
            printf('f2\n'),
            printf('%d\n', v2),
            printf('%d\n', v1),
            printf('%d\n', p2),
            printf('%d\n', p1),
            f3(30),
            0
        ) { where
            fun f3(p3 : integer) : integer = (
                { v3 = 3 },
                printf('f3\n'),
                printf('%d\n', v3),
                printf('%d\n', v2),
                printf('%d\n', v1),
                printf('%d\n', p3),
                printf('%d\n', p2),
                printf('%d\n', p1),
                f10(100),
                f20(200),
                0
            ) { where
                var v3 : integer
            };
            var v2 : integer
        };
        var v1 : integer;
        fun f20(p20 : integer) : integer = (
            printf('f20\n'),
            printf('%d\n', v1),
            printf('%d\n', p20),
